rules_version = '2';

// ============================================================
// Logic Systems – Firestore Security Rules
// ------------------------------------------------------------
// دلوقتي (لو مفيش rules متطبقة) أي حد عنده الـ apiKey بتاع
// المشروع (وهو ظاهر في الكود بالفعل، ده طبيعي في Firebase) يقدر
// يقرا/يكتب في القاعدة كلها من غير ما يسجل دخول أصلاً.
// الملف ده بيقفل الباب ده: محدش يقدر يعمل أي حاجة غير لو مسجل
// دخول فعليًا، وبيفرّق بين الأدمن والموظف حسب الإيميل.
// ============================================================

function isSignedIn() {
  return request.auth != null;
}

// الموظفين بيدخلوا بإيميل ثابت واحد (EMPLOYEE_EMAIL في الكود) —
// ده النظام القديم. أو بحساب مستقل ليه custom claim (role: "employee")
// لو اتفعّل نظام الحسابات المنفصلة عن طريق Cloud Function.
function isEmployeeAuth() {
  return isSignedIn()
    && (request.auth.token.email == "logic_system@gmail.com"
        || request.auth.token.role == "employee");
}

// الأدمن: إما حساب ليه مستند في customers/{uid} فيه active == true
// (النظام القديم)، أو حساب ليه custom claim role == "admin"
// (بعد ما تشغّل دالة setAdminClaim مرة واحدة، شوف functions/README.md).
function isAdmin() {
  return isSignedIn()
    && !isEmployeeAuth()
    && (request.auth.token.role == "admin"
        || (exists(/databases/$(database)/documents/customers/$(request.auth.uid))
            && get(/databases/$(database)/documents/customers/$(request.auth.uid)).data.active == true));
}

// أي مستخدم داخل فعليًا (أدمن أو موظف) — الموظف هنا "مسموح له
// بالدخول" مش بالضرورة "معتمد"، لأن ده حساب موحّد للكل.
// ملحوظة مهمة: طالما الموظفين بيستخدموا نفس الإيميل/الباسورد،
// مفيش طريقة تقنية 100% تمنع موظف "متطرود" من الدخول لحد ما
// تغيّر الباسورد الموحّد أو تحوّل لحسابات مستقلة لكل موظف.
function isStaff() {
  return isAdmin() || isEmployeeAuth();
}

// تنبيه: النظام لسه شغال بالحساب الموحّد للموظفين افتراضيًا (الواجهة
// الحالية في dashboard.html بتستخدمه). طالما كده، أي موظف مسجل
// دخول بيقدر يوصل لبيانات زي transactions لو حاول من برا الواجهة
// (DevTools)، لأن الـ Rules مقدرش تفرّق موظف عن موظف تاني بنفس
// الحساب. لما تفعّل الحسابات المنفصلة (functions/index.js -->
// createEmployeeAccount)، كل موظف هيبقى ليه custom claim مستقل،
// والـ isEmployeeAuth() فوق أصلاً بتتعرف عليه تلقائي من غير أي
// تعديل تاني هنا.

service cloud.firestore {
  match /databases/{database}/documents {

    // ---------- عملاء / أنظمة (customers) ----------
    // نفس الـ collection بتتخزن فيها كمان "مستندات تفعيل الأدمن"
    // (doc id = uid بتاع الأدمن). الأدمن يقدر يعمل كل حاجة،
    // الموظف يقدر يقرا ويعدّل بيانات العملاء بس مايقدرش يمسح
    // نهائي (فقط soft-delete عن طريق فيلد deleted).
    match /customers/{docId} {
      allow read: if isStaff();
      allow create: if isAdmin();
      allow update: if isStaff();
      allow delete: if isAdmin();
    }

    // ---------- طلبات/حسابات الموظفين ----------
    match /employeeRequests/{empId} {
      // الأدمن بس يقرا الليستة كاملة (بيانات موظفين + أرقام تليفون)
      allow read: if isAdmin();
      // أول مرة الموظف يدخل بيانات جهازه بيعمل طلب جديد بنفسه
      allow create: if isEmployeeAuth()
        && request.resource.data.status == "pending"
        && request.resource.data.totalMinutes == 0;
      // الأدمن يقدر يوافق/يرفض/يحدّث أي حاجة
      allow update: if isAdmin()
        // أو الموظف نفسه يحدّث بس الحقول الخاصة بجلسته (وقت الشغل
        // وآخر دخول) مش حالته أو بياناته الشخصية
        || (isEmployeeAuth()
            && request.resource.data.diff(resource.data).affectedKeys()
                 .hasOnly(["totalMinutes", "lastLoginAt", "updatedAt"]));
      allow delete: if isAdmin();
    }

    // ---------- إعدادات النظام ----------
    // ملحوظة: systemTypePassword لازم يتقرا من الموظف كمان (بوابة
    // إظهار بيانات الدخول)، فمينفعش نمنعه من القراءة، لكن التعديل
    // مقفول على الأدمن بس.
    match /settings/{docId} {
      allow read: if isStaff();
      allow write: if isAdmin();
    }

    // ---------- الاشتراكات/الماليات (transactions) ----------
    match /transactions/{docId} {
      allow read: if isStaff();
      allow create, update: if isStaff();
      allow delete: if isAdmin();
    }

    // ---------- الإشعارات ----------
    match /notifications/{docId} {
      allow read: if isStaff();
      allow create: if isStaff();
      allow update, delete: if isAdmin();
    }

    // ---------- شات الدعم الفني ----------
    match /supportChats/{chatId} {
      allow read, write: if isStaff();
    }

    // أي حاجة تانية غير كده: ممنوعة تمامًا
    match /{document=**} {
      allow read, write: if false;
    }
  }
}

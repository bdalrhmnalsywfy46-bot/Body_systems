const FIREBASE_PROJECT_ID = "body-system-dadaf";
const COLLECTIONS_TO_BACKUP = [

];
const DRIVE_FOLDER_NAME = "Logic Systems Backups";

// ---------- دالة التركيب (تشغّلها مرة واحدة بس) ----------
function setupDailyTrigger() {
  // بنشيل أي trigger قديم لنفس الدالة عشان منكررش الجدولة
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === "backupFirestore") {
      ScriptApp.deleteTrigger(t);
    }
  });
  ScriptApp.newTrigger("backupFirestore")
    .timeBased()
    .atHour(3)
    .everyDays(1)
    .inTimezone("Africa/Cairo")
    .create();

  // نجرب نعمل نسخة فورًا عشان نتأكد إن كل حاجة شغالة صح
  backupFirestore();
  Logger.log("تم التفعيل بنجاح — هيشتغل كل يوم الساعة 3 الصبح تلقائي");
}

// ---------- الدالة الرئيسية: تسجيل دخول + قراءة البيانات + حفظها ----------
function backupFirestore() {
  const props = PropertiesService.getScriptProperties();
  const email = props.getProperty("ADMIN_EMAIL");
  const password = props.getProperty("ADMIN_PASSWORD");

  if (!email || !password) {
    throw new Error("لازم تضيف ADMIN_EMAIL و ADMIN_PASSWORD في Script Properties الأول (خطوة 5 في الشرح فوق)");
  }

  const idToken = signIn(email, password);
  const backupData = {
    exportedAt: new Date().toISOString(),
    collections: {}
  };

  COLLECTIONS_TO_BACKUP.forEach(function (colName) {
    backupData.collections[colName] = fetchCollection(colName, idToken);
  });

  saveToGoogleDrive(backupData);
}

// ---------- تسجيل الدخول عن طريق Firebase Auth REST API ----------
function signIn(email, password) {
  const url = "https://identitytoolkit.googleapis.com/v1/accounts:signInWithPassword?key=" + FIREBASE_API_KEY;
  const response = UrlFetchApp.fetch(url, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify({ email: email, password: password, returnSecureToken: true }),
    muteHttpExceptions: true
  });
  const data = JSON.parse(response.getContentText());
  if (!data.idToken) {
    throw new Error("فشل تسجيل الدخول: " + (data.error ? data.error.message : "خطأ غير معروف") + " — تأكد إن ADMIN_EMAIL و ADMIN_PASSWORD صح");
  }
  return data.idToken;
}

// ---------- جلب كل مستندات collection معينة (مع pagination) ----------
function fetchCollection(colName, idToken) {
  const docs = {};
  let pageToken = null;

  do {
    let url = "https://firestore.googleapis.com/v1/projects/" + FIREBASE_PROJECT_ID +
      "/databases/(default)/documents/" + colName + "?pageSize=300";
    if (pageToken) url += "&pageToken=" + pageToken;

    const response = UrlFetchApp.fetch(url, {
      method: "get",
      headers: { Authorization: "Bearer " + idToken },
      muteHttpExceptions: true
    });
    const data = JSON.parse(response.getContentText());

    (data.documents || []).forEach(function (doc) {
      const idParts = doc.name.split("/");
      const docId = idParts[idParts.length - 1];
      docs[docId] = parseFirestoreFields(doc.fields || {});
    });

    pageToken = data.nextPageToken || null;
  } while (pageToken);

  return docs;
}

// ---------- تحويل صيغة Firestore REST (typed) لصيغة JSON عادية ----------
function parseFirestoreFields(fields) {
  const result = {};
  Object.keys(fields).forEach(function (key) {
    result[key] = parseFirestoreValue(fields[key]);
  });
  return result;
}

function parseFirestoreValue(value) {
  if (value.stringValue !== undefined) return value.stringValue;
  if (value.integerValue !== undefined) return parseInt(value.integerValue, 10);
  if (value.doubleValue !== undefined) return value.doubleValue;
  if (value.booleanValue !== undefined) return value.booleanValue;
  if (value.nullValue !== undefined) return null;
  if (value.timestampValue !== undefined) return value.timestampValue;
  if (value.mapValue !== undefined) return parseFirestoreFields(value.mapValue.fields || {});
  if (value.arrayValue !== undefined) {
    return (value.arrayValue.values || []).map(parseFirestoreValue);
  }
  return null;
}

// ---------- حفظ النسخة في Google Drive ----------
function saveToGoogleDrive(backupData) {
  const folders = DriveApp.getFoldersByName(DRIVE_FOLDER_NAME);
  const folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(DRIVE_FOLDER_NAME);

  const dateStamp = Utilities.formatDate(new Date(), "Africa/Cairo", "yyyy-MM-dd_HH-mm");
  const fileName = "backup-" + dateStamp + ".json";
  const jsonStr = JSON.stringify(backupData, null, 2);

  folder.createFile(fileName, jsonStr, MimeType.PLAIN_TEXT);
  Logger.log("تم الحفظ: " + fileName);
}
